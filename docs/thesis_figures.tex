\begin{figure}[t]
  \centering
  \includegraphics[width=0.86\textwidth]{fig_architecture_implemented}
  \caption[Implemented pipeline]{\textbf{Implemented two-stage pipeline.}
  \emph{Stage 1} — a Wavelet U-Net++ (Haar-DWT downsampling, dense nested skip
  connections) segments the tumour into three nested regions
  (ET~$\subseteq$~TC~$\subseteq$~WT); trained on BraTS 2021, applied to
  UPenn-GBM. \emph{Stage 2} — 51 radiomic features extracted from the predicted
  mask and the MRI are encoded (KAN \emph{or} MLP) with a region-token
  Transformer branch, optionally refined by a retrieval-augmented memory-bank
  GNN, and classified for MGMT methylation. Dashed blocks are ablation toggles
  or components exercised only on UPenn-GBM: BraTS 2021 ships no clinical
  metadata, so on the primary task the module is imaging-only and the clinical
  encoder and adaptive gate are inactive. The diagram reflects the code as run;
  the learned lesion/slice/global encoders of the original proposal remain
  unimplemented and are not drawn.}
  \label{fig:architecture}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=0.82\textwidth]{fig1_ablation_auc_forest}
  \caption[MGMT ablation: AUC by configuration]{\textbf{MGMT-methylation
  discrimination by configuration} on the primary cohort, BraTS 2021 ($n=577$;
  301 methylated, 52.2\%), imaging-only on the 51 radiomic features shared with
  UPenn-GBM, nested $5\times3$ cross-validation, 3 repeats. Points are mean
  held-out AUC; whiskers are 95\% confidence intervals. Every primary-cohort
  interval overlaps every other (mean 95\% CI half-width $\pm0.047$): the added
  components do not separate from the MLP baseline, and external transfer to
  UPenn-GBM is at chance.}
  \label{fig:forest}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=0.82\textwidth]{fig2_attribution}
  \caption[Component attribution]{\textbf{Component attribution via controlled
  one-factor contrasts.} \textbf{(a)} $\Delta$AUC contrasts, each isolating one
  change with the others held fixed; positive favours the more complex model.
  The grey band is the mean 95\% CI half-width ($\pm0.047$). No internal
  contrast clears it — KAN $-$ MLP ($-0.015$), $+$~Transformer ($+0.044$),
  $+$~memory-bank GNN ($+0.013$), and full module $-$ MLP baseline ($+0.007$)
  all lie inside the noise band — while external transfer to UPenn-GBM is
  negative ($-0.087$). \textbf{(b)} Graph reliance: the memory bank draws
  ${\approx}25\%$ of the fused representation from neighbours, yet this does not
  translate into an AUC gain in panel~(a).}
  \label{fig:attribution}
\end{figure}
